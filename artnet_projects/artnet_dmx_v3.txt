// --- DMX ENGINE TASK (CORE 0) ---
void DMXCoreTask(void *pvParameters) {
  for (;;) {
    digitalWrite(DMX_EN_PIN, HIGH);
    
    // --- CRITICAL SECTION: SWAP POINTER AMAN ---
    if (frameReady) {
      portENTER_CRITICAL(&mux); // Kunci Brankas!
      uint8_t* temp = readBuffer;
      readBuffer = writeBuffer;
      writeBuffer = temp;
      frameReady = false;
      portEXIT_CRITICAL(&mux);  // Buka Brankas!
    }
    
    // Break Signal (96000 baud)
    Serial2.updateBaudRate(96000);
    Serial2.write(0);
    Serial2.flush();
    
    // Data DMX (250000 baud)
    Serial2.updateBaudRate(250000);
    Serial2.write(0); // Start Code
    
    // Blackout if Loss Signal > 10 detik
    if (millis() - lastArtNetTime > 10000) {
      for (int i = 0; i < 512; i++) Serial2.write(0);
    } else {
      Serial2.write(readBuffer, 512); // Membaca dari Buffer Aktif
    }
    Serial2.flush();
    
    vTaskDelay(pdMS_TO_TICKS(22)); // Pertahankan 22 FPS yang sudah stabil
  }
}

// --- SETUP ---
void setup() {
  Serial.begin(115200);
  
  pinMode(DMX_EN_PIN, OUTPUT);
  digitalWrite(DMX_EN_PIN, LOW);
  Serial2.begin(250000, SERIAL_8N2, DMX_RX_PIN, DMX_TX_PIN);

  lcd.init();
  lcd.backlight();
  lcd.clear();

  // Bersihkan memori buffer saat booting awal
  memset(bufferA, 0, 512);
  memset(bufferB, 0, 512);

  xTaskCreatePinnedToCore(DMXCoreTask, "DMX_T", 2048, NULL, 2, &DMXTask, 0);

  prefs.begin("wifi_data", false);
  ssid = prefs.getString("ssid", "");
  pass = prefs.getString("pass", "");
  isAdvanced = prefs.getBool("isAdv", false);
  entUser = prefs.getString("entUser", "");
  entPass = prefs.getString("entPass", "");

  // Alur 1 & 2: Boot Screen -> Connecting Screen
  changeScreen(S_BOOT);
  changeScreen(S_CONNECTING);
  
  if (ssid == "") {
    // Alur 3 -> 4: WiFi Not Connect -> AP Mode
    changeScreen(S_WIFI_ERR);
    startAP();
  } else {
    // Mencoba Konek (Bisa sukses ke Alur 7, atau gagal ke Alur 11)
    connectWiFi();
  }
}

// --- MAIN LOOP ---
void loop() {
  handleLCDAnimation();
  server.handleClient();
  
  if (currentState == S_RUN) {
    int packetSize = Udp.parsePacket();
    if (packetSize) {
      char packetBuffer[530];
      Udp.read(packetBuffer, 530);
      
      // Art-Net Parsing dengan Cek Universe
      if (strncmp(packetBuffer, "Art-Net\0", 8) == 0 && packetBuffer[9] == 0x50) {
        
        uint8_t incomingUniverse = packetBuffer[14]; // Ambil ID Universe dari Paket
        
        if (incomingUniverse == myUniverse) {
          lastArtNetTime = millis();
          int length = (packetBuffer[16] << 8) | packetBuffer[17];
          if (length > 512) length = 512;
          
          // --- CRITICAL SECTION: TULIS DATA KE BUFFER ---
          portENTER_CRITICAL(&mux);
          
          memset(writeBuffer, 0, 512); // Cuci bersih buffer sebelum ditulis (Hapus Ghost Data)
          for (int i = 0; i < length; i++) {
            writeBuffer[i] = packetBuffer[18 + i];
          }
          frameReady = true; // Angkat bendera sinyal
          
          portEXIT_CRITICAL(&mux);
        }
      }
    }
    
    // Auto Reconnect / Disconnect Logic
    if (WiFi.status() != WL_CONNECTED) {
      if (wifiLostTime == 0) {
        wifiLostTime = millis();
        changeScreen(S_WIFI_DISCONN1); // Screen 10 (Reconnect <30s)
      }
      
      if (millis() - wifiLostTime > 30000) {
        // Screen 10 (Reboot >30s)
        changeScreen(S_WIFI_DISCONN2);
        prefs.clear();
        ESP.restart(); // Otomatis mengulang alur ke Setup Mode AP
      } else {
        WiFi.reconnect();
      }
    } else {
      if (wifiLostTime != 0) {
        // Jika reconnect sukses <30s, kembali ke Run
        changeScreen(S_CONNECTING);
        changeScreen(S_MOD_CONN2);
        changeScreen(S_RUN);
        wifiLostTime = 0; 
      }
    }
  }

  // AP Mode: Deteksi jika HP konek ke hotspot
  if (currentState == S_AP_MODE) {
    int currentDevices = WiFi.softAPgetStationNum();
    if (currentDevices > connectedDevices) {
      changeScreen(S_CONNECTING); // Screen 2
      changeScreen(S_DEV_CONN);   // Screen 5
      changeScreen(S_AP_MODE);    // Kembali ke Screen 4
    }
    connectedDevices = currentDevices;
  }
}


// --- WIFI & AP FUNCTIONS ---
void startAP() {
  WiFi.mode(WIFI_AP);
  WiFi.softAP("ARTNET-DMX");
  connectedDevices = 0;
  
  setupAPServer();
  server.begin();
  changeScreen(S_AP_MODE); // Screen 4
}

void connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(100);
  
  if (isAdvanced && entUser != "") {
    WiFi.begin(ssid.c_str(), WPA2_AUTH_PEAP, entUser.c_str(), entUser.c_str(), entPass.c_str());
  } else {
    WiFi.begin(ssid.c_str(), pass.c_str());
  }

  unsigned long startT = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startT < 10000) {
    delay(500);
  }

  if (WiFi.status() == WL_CONNECTED) {
    // Alur 7 -> 8: Module Connect 2 -> Run Mode
    changeScreen(S_MOD_CONN2);
    configTime(7 * 3600, 0, "pool.ntp.org");
    MDNS.begin("zzlight");
    Udp.begin(artNetPort);
    
    setupRunServer();
    server.begin();
    changeScreen(S_RUN);
  } else {
    // Alur 11 -> 4: WiFi Failed -> AP Mode
    changeScreen(S_WIFI_FAIL);
    startAP();
  }
}

// --- WEBSERVER ROUTING ---
void setupAPServer() {
  server.on("/", HTTP_GET, []() {
    server.sendHeader("Cache-Control", "no-cache, no-store, must-revalidate");
    server.sendHeader("Pragma", "no-cache");
    server.sendHeader("Expires", "-1");
    server.send(200, "text/html", AP_HTML);
  });

  server.on("/scan", HTTP_GET, []() {
    WiFi.scanDelete(); 
    int n = WiFi.scanNetworks(false, true); 
    String json = "[";
    for (int i = 0; i < n; ++i) {
      if (i > 0) json += ",";
      int dbm = WiFi.RSSI(i);
      int quality = (dbm <= -100) ? 0 : ((dbm >= -50) ? 100 : 2 * (dbm + 100));
      bool secure = WiFi.encryptionType(i) != WIFI_AUTH_OPEN;
      json += "{\"ssid\":\"" + WiFi.SSID(i) + "\",\"sec\":" + String(secure ? "true" : "false") + ",\"sig\":" + String(quality) + "}";
    }
    json += "]";
    server.send(200, "application/json", json);
  });
  
  server.on("/connect", HTTP_POST, []() {
    ssid = server.arg("ssid");
    pass = server.arg("pass");
    isAdvanced = server.arg("isAdv") == "1";
    entUser = server.arg("entUser");
    entPass = server.arg("entPass");
    
    prefs.putString("ssid", ssid);
    prefs.putString("pass", pass);
    prefs.putBool("isAdv", isAdvanced);
    prefs.putString("entUser", entUser);
    prefs.putString("entPass", entPass);
    
    server.send(200, "text/html", SAVED_HTML);
    
    // Screen 6 -> Reboot
    changeScreen(S_MOD_CONN1);
    ESP.restart();
  });
}

void setupRunServer() {
  server.on("/", HTTP_GET, []() {
    server.sendHeader("Cache-Control", "no-cache, no-store, must-revalidate");
    server.sendHeader("Pragma", "no-cache");
    server.sendHeader("Expires", "-1");
    server.send(200, "text/html", RUN_HTML);
  });

  server.on("/status", HTTP_GET, []() {
    struct tm ti;
    String dateStr = "Loading...";
    String timeStr = "Loading...";
    if (getLocalTime(&ti, 50)) {
      char dBuf[50], tBuf[20];
      sprintf(dBuf, "%s, %02d %s %d", dayNames[ti.tm_wday].c_str(), ti.tm_mday, monthNames[ti.tm_mon].c_str(), ti.tm_year + 1900);
      sprintf(tBuf, "%02d:%02d:%02d", ti.tm_hour, ti.tm_min, ti.tm_sec);
      dateStr = String(dBuf);
      timeStr = String(tBuf);
    }

    unsigned long dmxDelta = millis() - lastArtNetTime;
    String dmxStat = "Loss Signal";
    if (dmxDelta < 1000) dmxStat = "Active";
    else if (dmxDelta < 10000) dmxStat = "No Signal";

    String json = "{";
    json += "\"date\":\"" + dateStr + "\",";
    json += "\"time\":\"" + timeStr + "\",";
    json += "\"ip\":\"" + WiFi.localIP().toString() + "\",";
    json += "\"wifi\":\"" + WiFi.SSID() + "\",";
    json += "\"status\":\"" + String(WiFi.status() == WL_CONNECTED ? "Connected" : "Disconnected") + "\",";
    json += "\"strength\":\"" + String(WiFi.RSSI()) + "\",";
    json += "\"dmx\":\"" + dmxStat + "\"";
    json += "}";
    
    server.send(200, "application/json", json);
  });

  server.on("/reset", HTTP_POST, []() {
    server.send(200, "text/plain", "OK");
    prefs.clear();
    
    // Screen 9 -> Reboot
    changeScreen(S_WIFI_RST);
    ESP.restart();
  });

  server.on("/reboot", HTTP_POST, []() {
    server.send(200, "text/plain", "OK");
    delay(500);
    ESP.restart();
  });
}

// --- LCD CONTROLLER ---
void changeScreen(ScreenState newState) {
  // Hanya simpan state jika bukan state non-blocking
  if(newState != S_WIFI_DISCONN1) {
    currentState = newState;
  }
  lcd.clear();
  
  switch(newState) {
    case S_BOOT: // Screen 1
      for(int i=0; i<7; i++) { lcd.setCursor(5+i, 0); lcd.print("ZZLIGHT"[i]); delay(150); }
      for(int i=0; i<7; i++) { lcd.setCursor(5+i, 1); lcd.print("PROJECT"[i]); delay(150); }
      delay(900); // Penyeimbang agar total durasi sekitar 3 detik
      break;
      
    case S_CONNECTING: // Screen 2
      lcd.setCursor(0, 0); lcd.print("Connecting...   ");
      lcd.setCursor(0, 1); lcd.print("                ");
      delay(3000);
      break;
      
    case S_WIFI_ERR: // Screen 3
      lcd.setCursor(0, 0); lcd.print("WiFi Not Connect");
      lcd.setCursor(0, 1); lcd.print("Connecting...   ");
      delay(3000);
      break;
      
    case S_AP_MODE: // Screen 4
      lcd.setCursor(0, 0); lcd.print("ARTNET-DMX      ");
      lcd.setCursor(0, 1); lcd.print("192.168.4.1     ");
      break;
      
    case S_DEV_CONN: // Screen 5
      lcd.setCursor(0, 0); lcd.print("Device Connect! ");
      lcd.setCursor(0, 1); lcd.print("Loading...      ");
      delay(3000);
      break;
      
    case S_MOD_CONN1: // Screen 6
      lcd.setCursor(0, 0); lcd.print("Module Connect! ");
      lcd.setCursor(0, 1); lcd.print("Rebooting...    ");
      delay(3000);
      break;
      
    case S_MOD_CONN2: // Screen 7
      lcd.setCursor(0, 0); lcd.print("Module Connect! ");
      lcd.setCursor(0, 1); lcd.print("Loading...      ");
      delay(3000);
      break;
      
    case S_RUN: // Screen 8
      animStep = 0;
      lastAnimTime = millis();
      lcd.setCursor(0, 0); lcd.print("ARTNET-DMX      ");
      lcd.setCursor(0, 1); 
      {
        String ipStr = WiFi.localIP().toString();
        // Spasi pengisi untuk menghilangkan karakter sisa (flicker-free)
        while(ipStr.length() < 16) ipStr += " ";
        lcd.print(ipStr);
      }
      break;
      
    case S_WIFI_RST: // Screen 9
      lcd.setCursor(0, 0); lcd.print("WiFi Reset!     ");
      lcd.setCursor(0, 1); lcd.print("Rebooting...    ");
      delay(3000);
      break;
      
    case S_WIFI_DISCONN1: // Screen 10 (Reconnect <30s)
      lcd.setCursor(0, 0); lcd.print("WiFi Disconnect!");
      lcd.setCursor(0, 1); lcd.print("Reconnecting... ");
      break;
      
    case S_WIFI_DISCONN2: // Screen 10 (Reboot >30s)
      lcd.setCursor(0, 0); lcd.print("WiFi Disconnect!");
      lcd.setCursor(0, 1); lcd.print("Rebooting...    ");
      delay(3000);
      break;
      
    case S_WIFI_FAIL: // Screen 11
      lcd.setCursor(0, 0); lcd.print("WiFi Failed!    ");
      lcd.setCursor(0, 1); lcd.print("Wrong Password..");
      delay(3000);
      break;
  }
}

// Animasi loop untuk S_RUN (6 Langkah yang Halus Tanpa Kedip)
void handleLCDAnimation() {
  if (currentState == S_RUN) {
    if (millis() - lastAnimTime > 500) {
      lastAnimTime = millis();
      animStep++;
      if (animStep > 5) animStep = 0;
      
      lcd.setCursor(10, 0); 
      if (animStep == 0) {
        lcd.print("      "); // Kosong (Langkah 1)
      } else {
        lcd.print(" "); 
        for(int i=1; i<=animStep; i++) lcd.print(">");
        for(int i=animStep+1; i<=5; i++) lcd.print(" ");
      }
    }
  }
}
